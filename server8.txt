const express = require('express');
const app = express();
app.use(express.json());
app.use((req, res, next) => {
  res.header('Access-Control-Allow-Origin', '*');
  res.header('Access-Control-Allow-Headers', 'Content-Type');
  res.header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS');
  if (req.method === 'OPTIONS') return res.sendStatus(200);
  next();
});
app.get('/', (req, res) => { res.json({ ok: true, service: 'DOLAI Backend AI' }); });
async function getReply(message, key) {
  if (!key) return 'الخادم يخدم، بصح المفتاح مازال ما تحطش في Render.';
  try {
    // 1. نسقسي Google واش هي الموديلات اللي تخدم
    const listR = await fetch('https://generativelanguage.googleapis.com/v1beta/models?key=' + key);
    const listData = await listR.json();
    const models = (listData.models || []).filter(m => (m.supportedGenerationMethods || []).includes('generateContent')).map(m => m.name.replace('models/', ''));
    // نفضل الفلاش، من بعد أي موديل Gemini
    const preferred = models.find(n => n.includes('flash')) || models.find(n => n.includes('gemini')) || models[0];
    if (!preferred) return 'Google ما رجع حتى موديل. المفتاح يخدم بصح ما كاش موديل متاح.';
    // 2. نجرب بالموديل اللي لقيناه
    const r = await fetch('https://generativelanguage.googleapis.com/v1beta/models/' + preferred + ':generateContent?key=' + key, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ contents: [{ parts: [{ text: 'You are DOLAI, a helpful Algerian AI assistant. Answer in simple Algerian dialect. User question: ' + message }] }] })
    });
    const data = await r.json();
    const reply = (data && data.candidates && data.candidates[0] && data.candidates[0].content && data.candidates[0].content.parts && data.candidates[0].content.parts[0] && data.candidates[0].content.parts[0].text);
    if (reply) return reply;
    return 'Google قال: ' + (data.error ? data.error.message : 'ما قدرتش نجاوب');
  } catch (e) { return 'صرات مشكلة: ' + e.message; }
}
app.get('/api/chat', async (req, res) => {
  const reply = await getReply(req.query.message || 'هاي', process.env.GEMINI_API_KEY);
  res.json({ reply });
});
app.post('/api/chat', async (req, res) => {
  const reply = await getReply((req.body && req.body.message) || '', process.env.GEMINI_API_KEY);
  res.json({ reply });
});
app.get('/api/models', async (req, res) => {
  const key = process.env.GEMINI_API_KEY;
  if (!key) return res.json({ error: 'المفتاح ما كاينش' });
  const r = await fetch('https://generativelanguage.googleapis.com/v1beta/models?key=' + key);
  const data = await r.json();
  res.json(data);
});
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => console.log('DOLAI AI running on ' + PORT));
